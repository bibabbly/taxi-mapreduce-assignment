hdfs dfsadmin -report

# JOB 1 - HOURLY DEMAND  (application_1788043663698_0003)
hdfs dfs -rm -r /taxi_project/output/hourly
hadoop jar C:\hadoop\share\hadoop\tools\lib\hadoop-streaming-3.3.6.jar -files "mappers/mapper_hourly.py,reducers/reducer_hourly.py" -input /taxi_project/input/raw -output /taxi_project/output/hourly -mapper "python mapper_hourly.py" -reducer "python reducer_hourly.py"
hdfs dfs -cat /taxi_project/output/hourly/part-00000

# JOB 2 - REVENUE BY PICKUP ZONE  (application_1788043663698_0004)
hdfs dfs -rm -r /taxi_project/output/revenue
hadoop jar C:\hadoop\share\hadoop\tools\lib\hadoop-streaming-3.3.6.jar -files "mappers/mapper_revenue.py,reducers/reducer_revenue.py" -input /taxi_project/input/raw -output /taxi_project/output/revenue -mapper "python mapper_revenue.py" -reducer "python reducer_revenue.py"
hdfs dfs -cat /taxi_project/output/revenue/part-00000

# JOB 3 - TOP 10 BY REVENUE, STAGE 2  (application_1788043663698_0005)
# Reads the HDFS output of Job 2. Single reducer forces a global ordering.
hadoop jar C:\hadoop\share\hadoop\tools\lib\hadoop-streaming-3.3.6.jar -D mapreduce.job.reduces=1 -files "mappers/mapper_topn.py,reducers/reducer_topn.py" -input /taxi_project/output/revenue -output /taxi_project/output/top10_revenue -mapper "python mapper_topn.py" -reducer "python reducer_topn.py"
hdfs dfs -cat /taxi_project/output/top10_revenue/part-00000

# -files list must be quoted or cmd.exe splits on the comma
# NodeManager local dirs need winutils chmod 755
